\textbf{Complexity:~\cite{ben2013quantum}}
\begin{itemize}
    \item Unital noise: compute for ${O}{(\text{log}(n))}$ steps.  
    \item Dephasing noise: compute on $O(n^a)$ qubits for ${O}(n^{1-a})$ steps.
    \item   Amplitude damping noise: compute for an exponential number of time steps
\end{itemize}

\section{Single qubit incoherent noise}

\begin{frame}{Incoherent noise }
    \begin{lemma}[~\cite{ruskai2002analysis}]
    Any single-qubit noise channel can be decomposed into coherent and incoherent parts
\[
\mathcal{N}(\rho) = U \Lambda(V \rho V^\dagger)U^\dagger,
\]
where $U,V$ are unitary gates and  \(\Lambda\left(\frac{1}{2}[\mathbb{I} + \mathbf{w} \cdot \vec \sigma]\right) = \frac{1}{2}[\mathbb{I} + (\mathbf{t} + \Gamma\mathbf{w}) \cdot  \vec\sigma]\). Here, \(\Gamma\) is a diagonal matrix with real diagonal entries $(\Gamma_1,\Gamma_2,\Gamma_3)$ and $\mathbf{t} = (t_1,t_2,t_3)$ is a real vector. 
\end{lemma}
%%%%%%%%%%%%%%%%%%%%%%%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{itemize}
    \item It is recently aware that coherent errors can generate high-magic resource states for universal quantum computing. ~\cite{niroula2024phase,turkeshi2024coherent}
    \item In the case of $\mathbf{t} = 0$, this channel reduced to the Pauli noise. When \(\mathbf{t} \neq 0\), the quantum channel is non-unital.
\end{itemize}
\end{frame}

\begin{frame}{Back-propagation property of incoherent channel}
    \begin{theorem}\label{theo:non-unital_back-propagation}
Conjugated channel $\Lambda^\dagger(\cdot)$ operates on Pauli operators  $\{\sigma_1,\sigma_2,\sigma_3\}$ as a scaling part $\mathbb{S}$ followed by a shifting part $\mathbb{T}$ such that
   \begin{align}
       \sigma_i &\xrightarrow{\mathbb{S}(\cdot)} s_i \sigma_i\\
       &\xrightarrow{\mathbb{T}(\cdot)} s_i\cdot \left[(1 - p_i)(\text{sign}~\Gamma_i \cdot \sigma_i) + p_i(\text{sign}~t_i\cdot\mathbb{I})\right]\\
       &= \Lambda^\dagger(\sigma_i),
   \end{align}
   where $s_i = |\Gamma_i| + |t_i| \leq 1, p = {|t_i|}{(|\Gamma_i| + |t_i|)}^{-1} \in [0,1]$. 
\end{theorem}
\end{frame}

\section{Simulation of Pauli average}
\begin{frame}[allowframebreaks]{Free on no-magic circuit}
   \begin{figure}
       \centering
       \includegraphics[width=0.5\linewidth]{Presentation/pre2_CliffordAD.png}
   \end{figure}
\newpage


\begin{theorem}
   Consider a fixed Clifford NISQ circuit composed of $\ket{0}^{\otimes n}$ inputs, Clifford gate layers, and an amplitude-damping channel with parameter $p < 1/2$. To estimate the Pauli average within an additive constant error $\varepsilon$ for all circuits of this kind, 
   \begin{itemize}
       \item the complexity of the Pauli path truncation algorithm must exhibit sub-exponential growth, with a bound of $O(n^{-1} e^{H(p) \cdot n})$,where $H(p) = -p\ln p - (1-p)\ln (1-p)$. 
       \item the randomized algorithm estimates its Pauli average within constant error and $\text{Var}\leq 1$.
   \end{itemize}
\end{theorem}

\end{frame}

\begin{frame}[allowframebreaks]{Add T gates: Gate sampling algorithm}

     \begin{align}
     U^\dagger \sigma_i U= \sum_{r\in R_i} \sqrt{q_{i,r}} \text{sign}_{i,r} \sigma_r,\quad  \sum_i q_i = 1.
 \end{align}

\begin{theorem}[Gate sampling method] \label{theo:gate-sampling-method}
The gate sampling algorithm have the expectation value and upper bound for variance:
\begin{align}
    \mathbb{E}  = \text{tr}( P^{(0)} C[\ket{0}\!\bra{0}^{\otimes n}] ).
\end{align}
\color{red}
\begin{align}
    \text{Var}&\leq \exp\left[ \frac{1}{2} M^{(1/2)}[C_N^\dagger(O)]  + M^{(0)}[C_U^\dagger(O)] \right].
\end{align}
\color{black}
where $\mathcal{M}^{(\alpha)}({\Lambda}^\dagger[O])$ represents the $\alpha$-operator stabilizer entropy. $C_U$ denotes the unitary part of the circuit $C$, and $C_N$ denotes the incoherent-noise part of the circuit.
\end{theorem}

\end{frame}
\begin{frame}{Free on Sparse T gates circuit}
 $T^\dagger \sigma_3 T = \sigma_3$,$T^\dagger \sigma_1 T = ({1}/{\sqrt{2}}) (\sigma_1-\sigma_2 )$, and $T^\dagger \sigma_2 T =  ({1}/{\sqrt{2}}) (\sigma_1 + \sigma_2)$.
 \begin{figure}[htbp!]
    \centering
    \includegraphics[width=0.4\linewidth]{Presentation/Gate_samp_example.png}
\end{figure}
\begin{theorem}\label{theo:sparse_CT}
    Clifford+T circuit \( C \) with depolarizing noise layers \( \Lambda(\rho) = (1 - p)\rho +({p}/{2}) \mathbb{I} \), the variance of the algorithm is $\text{Var} \leq \sum_{i \in N_d} |f(i)|^2.$\\
For a 2-dimensional, \( (Q, k) \)-sparse, localized circuit, if $2^Q(1 - p)  < 1$, the variance of the gate sampling algorithm satisfies \( \text{Var} \leq \text{poly}(n) \). Further more, adds amplitude damping noise into this circuit will
never increase the variance.
\end{theorem}
\end{frame}
