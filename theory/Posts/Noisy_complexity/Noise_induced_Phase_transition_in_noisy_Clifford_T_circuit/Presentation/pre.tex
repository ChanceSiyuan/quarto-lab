\begin{frame}[allowframebreaks]{Percolation simulation}
\textbf{Percolation on noisy-IQP~\cite{rajakumar2025polynomial}}
\begin{theorem}
    Given an $k$-local NISQ-IQP circuit of depth $d$ on $n$ qubits with amplitude dampling noise. Denote $p = p_z + \text{min}(p_x,p_y)$. If $d > O(p^{-1} \text{log}(k p^{-1}))$, there exists a randomized classical algorithm that samples $Q$ with $\|Q-P\|_{TVD} \leq \varepsilon$ within time $T \leq O(d ~\text{poly}(n/\varepsilon))$ .
\end{theorem}
\begin{lemma}
    Any Pauli noise channel can be decomposed as
    \begin{align}
        \mathcal{N}_{i}'(\rho) = (1 - 2p)\mathcal{N}_i^{(1)}(\rho) + 2p\mathcal{N}_i^{(2)} \circ \Lambda_z(\rho),
    \end{align}
    where $p = p_Z + \min(p_X,p_Y)$ .  $N^{(2)}_i$ and  $N^{(1)}_i$ are Pauli noise. $\Lambda_z  = ({1}/{2})\rho+({1}/{2})Z\rho Z$ 
\end{lemma}

        \item $\Lambda_z(\rho)$  decoheres $\ket{+}$ state into uniform state ensemble $\{\ket{0}\!\bra{0}, \ket{1}\!\bra{1}\}$. 
    \item   $\{\ket{0}\!\bra{0}, \ket{1}\!\bra{1}\}$ will not entangled with the other $\ket{+}$  states under the operation of phase gates. 
\end{itemize}

\section{Simulation of circuit amplitude}
\begin{frame}[allowframebreaks]{Diagonalization of T-gadget circuit}
\begin{theorem}[Noisy reduction to graph states]
\label{lemma:Noisy_reduction}
Use $\text{Cl}$ to represent Clifford gate and $U[\rho]:= U \rho U^\dagger$ to imply unitary operation on mixed state.
 $\mathcal{N}$ is noise channel that has the form
 \begin{align}
    \mathcal{N}\left(\frac{1}{2}[\mathbb{I} + \mathbf{w} \cdot \vec \sigma]\right) = \frac{1}{2}[\mathbb{I} + (\mathbf{t} + \Gamma\mathbf{w}) \cdot  \vec\sigma] .
 \end{align}
In the absence of noise, Clifford operation on stabilizer state $\ket{S}$ can be equivalently realized by a set of $CZ$ gates $G$  followed by local Clifford gates $L$,  $\text{Cl} \ket{S}= LG \ket{S}.$ Further more, this identity is still satisfied under Pauli noise:\begin{align}
    \text{Cl} \circ \mathcal{N}[\ket{S}\!\bra{S}] = LG \circ \mathcal{N}[\ket{S}\!\bra{S}].
\end{align} 
\end{theorem}

\begin{theorem}[Diagonalzation of Pauli-noise Clifford circuit]
\label{theo:diagonalzation}
Consider NISQ circuit with $d$ layer of Clifford gates  $\{\text{Cl}_i\}_{i =1}^d$  and $T$ gates. After replacing $T$ gates with $T$-gadget and decompose Clifford operation $\{\text{Cl}_i\}_{i =1}^d$ into $CZ$ gates $\{G_i\}_{i =1}^d$ as proved in Lemma~\ref{lemma:Noisy_reduction} , we get the circuit in Figure~\ref{fig:Percolated_stabilizer_state}(b). Those transformed gates and noisy channel can be calculated from NISQ circuit through the relation below
\begin{align}
    &G_j G_{j-1}\cdots G_1=D_j^\dagger,
    \text{where~}\text{Cl}_j^\dagger \cdots \text{Cl}_1^\dagger \ket{+}^{\otimes n} = L_j^\dagger D_j^\dagger \ket{+}^{\otimes n}   ~\forall j = 1, \cdots, d.  \\
     &P_{j}= A_j Z A_j^\dagger,\\
     &\text{where~} A_d = L_d, A_j =   (G_{j+1} \cdots G_d )L_d(\text{Cl}_{j+1} \cdots\text{Cl}_d)^\dagger ~\forall j = 1, \cdots,d-1. \\
    &\mathcal{N}_j^{'}(\rho) = L_j\circ \mathcal{N}_j \circ L_j^\dagger(\rho) .
\end{align}
\end{theorem}
\end{frame}


\begin{frame}{Measurement Operation on graph}
    \begin{lemma}[measurements on graph]\label{lemma:Pauli_measurement_on_graph}
    The graph state $\ket{G}$ undergoes a Pauli measurement at vertex $a$, resulting in a new graph state $\ket{G'}$ that is locally Clifford equivalent. Specifically:
\begin{enumerate}
    \item After a $Z$-measurement, the graph is modified by deleting all edges incident to vertex $a$.
    \item For a $Y$-measurement, the neighborhood of $a$ is flipped, and then all edges incident to $a$ are removed.
    \item For an $X$-measurement, choose a neighboring vertex $b$ of $a$. First, flip the neighborhood of $b$, then flip the neighborhood of $a$, followed by removing the edges incident to $a$, and finally flip the neighborhood of $b$ again.
\end{enumerate}
\end{lemma}
\end{frame}

\begin{frame}{Combine together: generate random graphs}
\begin{enumerate}
    \item Initialized with $n$ original vertexes. 
    \item For each layer of gates:
    \begin{itemize}
        \item For Clifford gates, change the connections of $n$ original vertexes.
        \item For each T-gates on qubits $i$, add a new vertex and connect it with $i$-th original vertex or its neighbors.
        \item For every local Pauli noise on $i$-th qubits, remove $i$-th original vertex and edge inserted on this vertex.
    \end{itemize}
    \item For the measurement on $\ket{0^{\otimes n}}$, randomly \textbf{remove} or \textbf{reversely remove} all original vertexes.
    \item Get a graph on the rest $t$ vertexes. 
    \item Calculate the inner product between graph state and product state---MBQC simulation.
\end{enumerate}
\begin{theorem}\label{theo:variance_of_percolation_sampling}
    The algorithm samples from graph states along with local Clifford gates $\{\ket{G_i},L_i\}_i$ ,then each of them satisfied $|\bra{G_i}L_i(H\ket{T})^{\otimes n}|^2 < 2^{-t}$.   
\end{theorem}
\end{frame}

\begin{frame}[allowframebreaks]{MBQC simulation based on entanglement~\cite{van2007classical}}
\begin{definition}
    Given an \( n \)-qubit state \( \ket{\psi} \), its vertices are divided using a tree bisection on the \( T \)-tree. The entanglement width is defined as
\[
E_{\mathrm{wd}}(\ket{\psi}) := \min_T \max_{e \in T} E_{A_T^e, B_T^e}(\ket{\psi}),
\]
where \( E_{A_T^e, B_T^e}(\ket{\psi}) \) is the entanglement entropy for the partition \( (A_T^e, B_T^e) \) of the state \( \ket{\psi} \).
\end{definition}
\begin{theorem}
Given an \( n \)-qubit state \( \ket{\psi} \), and a \( n \)-qubit tensor tree graph \( T \) . There exists a tree tensor network representation with the maximum tensor dimension \( D \) of the tensor satisfies $\log_2 D = \max_{e \in T} \chi_{A_T^e, B_T^e}(\ket{\psi})$.
\end{theorem}
\newpage
\begin{theorem}
For graph states, determining whether the rank-width is smaller than $k$ is a problem in $\textbf{P}$. There exists an algorithm with a time complexity of $O(n^3)$ that either certifies $\text{rwd}(G) > k$ or outputs a subcubic tree $T^*$ such that:
\[
\operatorname{rwd}(G) \leq \max_{e \in T^*} \operatorname{rank}_{\mathbb{F}_2} \Gamma(A_{T^*}^e, B_{T^*}^e) = 3k - 1.
\]
\end{theorem}
\begin{theorem}
    If there exists a tree such that the graph rank width of an $n$-qubit graph state $\ket{G}$ under its bipartition grows logarithmically with $n$, then $\ket{G}$ can be efficiently transformed into a tree tensor network with a bond dimension smaller than $\text{rwd}(G)$. Consequently,MBQC on $\ket{G}$ can be efficiently simulated classically.
\end{theorem}
\end{frame}
