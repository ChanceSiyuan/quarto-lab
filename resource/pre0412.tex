\section{How to connect coherence and magic?}
\begin{frame}{Generalized Clifford}
\begin{definition}[matrix basis Ring]
    A matrix basis ring is a set of linear independent matrix basis $\mathcal{B}=\{{B}_j\}_{j = 1}^{n^2} \cup \{0\}$ such that ${B}_j \cdot{B}_k \in \mathcal{B}, B_j^\dagger \in \mathcal{B}$. The unitary transformation such that  $U^\dagger B_iU = \alpha_i B_{\pi (i)}$ is call the generalized Clifford of $\mathcal{B}$, which we denoted as $\mathcal{C}(\mathcal{B})$. $U$ is a Pauli components in $\mathcal{C}(\mathcal{B})$ iff $U^\dagger B_iU = \alpha_i B_i$, which we denote $\mathcal{P}(\mathcal{B})$. 
\end{definition}
\begin{itemize}
    \item Pauli basis:
    \begin{itemize}
        \item $\mathcal{B}_{\text{Pauli}}:=\{\pm i,\pm1\} \times \{I,Z,X,Y\}^n \cup \{0\}$
        \item $\mathcal{C}(\mathcal{B}_{\text{Pauli}}) \sim \{CNOT,S,H\} $. 
        \item  $\mathcal{P}(\mathcal{B}_{\text{Pauli}})  =\mathcal{B}_{\text{Pauli}} $.
    \end{itemize}
    \item Computational basis:
    \begin{itemize}
        \item  $\mathcal{B}_{\text{comp}} = \{\pm i, \pm 1\} \times \{\ket{i}\!\bra{j}\}_{i,j = 0,1}^{\otimes n}\cup \{0\}$
        \item $\mathcal{C}(\mathcal{B}_{\text{comp}})\sim \sum_{j \in \{0,1\}^n}e^{i\phi(j) } \ket{\pi(j)}\!\bra{j}$,
        \item   $\mathcal{P}(\mathcal{B}_{\text{comp}}) = \sum_{j \in \{0,1\}^n}e^{i\phi(j) } \ket{j}\!\bra{j}  \neq \mathcal{B}_{\text{comp}} $ 
    \end{itemize}
\end{itemize}

\end{frame}
\begin{frame}{Generalized Clifford hierachy}
    \begin{lemma}
     Take $P \in \mathcal{P}(\mathcal{B}), \text{Cl} \in \mathcal{C}(\mathcal{B})$, then $\text{Cl}^\dagger P\text{Cl} \in \mathcal{P}(\mathcal{B})$.
\end{lemma}
 
\begin{definition}[Generalized Clifford Hierachy]
     The $k$-th hierarchy of generalized Cliffords are  $\mathcal{C}_{k}(\mathcal{B}):=\{\text{Cl}_k|\text{Cl}_k^\dagger P\text{Cl}_k \in \mathcal{C}_{k-1}(\mathcal{B}), \forall P \in \mathcal{P}(\mathcal{B}) \}$, where $\mathcal{C}_{1}(\mathcal{B}) := \mathcal{C}(\mathcal{B})$.
\end{definition}
\begin{itemize}
    \item In the case of $\mathcal{B}_{\text{comp}}$, it states that $CZ,CCZ$ can be moved before $CNOT,CY$ with certain phase gate correction.
    \item Conjugates of $H$ will transform $Z,CZ$ or $CCZ$ into $X,CNOT$ and Toffoli gates.
    \item \color{red} $H \in \mathcal{C}_2(\mathcal{B}_{\text{comp}})$? \color{black}
\end{itemize}
\end{frame}
\begin{frame}{Equivalence between all possible Clifford structure}
    \begin{conjecture}
         The vectorization of $\mathcal{B}$ with real qubit $a$ and visual qubit $b$ is identical under an unitary transformation within vectorization space.
         Specifically, the unitary transformation $CS_{ba}^\dagger H_bCNOT_{ba}$ relates $\mathcal{B}_{\text{Pauli}}$ to $\mathcal{B}_{\text{comp}}$. 
    \end{conjecture}
        \begin{align*}
    &\{I,Z,X,Y\} \\
    &\xrightarrow{\mathcal{V}}\{\frac{\ket{00}_{ab}+\ket{11}_{ab}}{\sqrt{2}},\frac{\ket{00}_{ab}-\ket{11}_{ab}}{\sqrt{2}},\frac{\ket{01}_{ab}+\ket{10}_{ab}}{\sqrt{2}},\frac{-i\ket{01}_{ab}+i\ket{10}_{ab}}{\sqrt{2}}\}\\
    &\xrightarrow{CS_{ba}^\dagger H_bCNOT_{ba}}\{\ket{0}_a\ket{0}_b, \ket{0}_a\ket{1}_b, \ket{1}_a\ket{0}_b,\ket{1}_a\ket{1}_b\},\\
    &\xrightarrow{\mathcal{V}^{-1}}\{\ket{0}\!\bra{0}, \ket{0}\!\bra{1}, \ket{1}\!\bra{0},\ket{1}\!\bra{1}\}
\end{align*}
    \begin{itemize}
        \item Is coherence really identical with magic?
    \end{itemize}
\end{frame}
